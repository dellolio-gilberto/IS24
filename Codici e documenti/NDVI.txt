//MAIN INIZIATO
function Main(aoi, roi){
  
  var col = ee.ImageCollection('MODIS/061/MOD13A2').select('NDVI');

  var start = startDate.getValue();
  var end = endDate.getValue();
  var country = co.getValue();

  var mask = ee.FeatureCollection('USDOS/LSIB_SIMPLE/2017')
    .filter(ee.Filter.eq('country_co', country));
  //wld_rgn
  //country_co

  // aggiungi il doy
  col = col.map(function(img) {
    var doy = ee.Date(img.get('system:time_start')).getRelative('day', 'year');
    return img.set('doy', doy);
  });
  //collezione doy
  var distinctDOY = col.filterDate(start,end);

  // filtra le immagini per doy 
  var filter = ee.Filter.equals({leftField: 'doy', rightField: 'doy'});

  // join
  var join = ee.Join.saveAll('doy_matches');

  // FeatureCollection -> ImageCollection
  var joinCol = ee.ImageCollection(join.apply(distinctDOY, col, filter));

  // Riduttore mediano
  var comp = joinCol.map(function(img) {
    var doyCol = ee.ImageCollection.fromImages(
      img.get('doy_matches')
    );
  return doyCol.reduce(ee.Reducer.median())
    .copyProperties(img, ['system:time_start']);
});

var visParams = {
  min: 0.0,
  max: 9000.0,
  palette: [
    'FFFFFF', 'CE7E45', 'DF923D', 'F1B555', 'FCD163', '99B718', '74A901',
    '66A000', '529400', '3E8601', '207401', '056201', '004C00', '023B01',
    '012E01', '011D01', '011301'
  ],
};

// Parametri di visualizzazione della GIF
  var rgbVis = comp.map(function(img) {
    return img.visualize(visParams).clip(mask);
  });
  
  return rgbVis;
}

function gifp(aoi){
  
  drawingTools.setShape(null);

  // pannello visibile alla prima geometria
  if (!chartPanel.style().get('shown')) {
    chartPanel.style().set('shown', true);
  }

  // Drawing mode off
  drawingTools.setShape(null);
  
  var mapScale = Map.getScale();
  var scale = mapScale > 5000 ? mapScale * 2 : 5000;
  
  var roi = aoi;
  // GIF parametri di output
  var gifParams = {
    'region': roi,
    'dimensions': 600,
    'crs': 'EPSG:3857',
    'framesPerSecond': 10,
    'format': 'gif'
  };
  
  var thumb = ui.Thumbnail(Main(aoi, roi),gifParams);
  // Rimpiazza il grafico 
  chartPanel.widgets().reset([thumb]);
}

drawingTools.onDraw(ui.util.debounce(gifp, 500));
drawingTools.onEdit(ui.util.debounce(gifp, 500));

function chartNdviTimeSeries() {
  if (!chartPanelC.style().get('shown')) {
    chartPanelC.style().set('shown', true);
  }

  var aoi = drawingTools.layers().get(0).getEeObject();

  drawingTools.setShape(null);

  var mapScale = Map.getScale();
  var scale = mapScale > 5000 ? mapScale * 2 : 5000;
  
  var start = startDate.getValue();
  var end = endDate.getValue();

  var chartNDVI = ui.Chart.image
                  .seriesByRegion({
                    imageCollection: ee.ImageCollection('MODIS/061/MOD13A2'),
                    regions: aoi,
                    reducer: ee.Reducer.mean(),
                    band: 'NDVI',
                    scale: scale,
                    xProperty: 'system:time_start'
                  })
                  .setOptions({
                    titlePostion: 'none',
                    legend: {position: 'none'},
                    hAxis: {title: 'Date'},
                    vAxis: {title: 'NDVI (x1e4)'},
                    series: {0: {color: '23cba7'}}
                  });

  chartPanelC.widgets().reset([chartNDVI]);
}

drawingTools.onDraw(ui.util.debounce(chartNdviTimeSeries, 500));
drawingTools.onEdit(ui.util.debounce(chartNdviTimeSeries, 500));
