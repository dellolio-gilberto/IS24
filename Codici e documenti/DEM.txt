function clip(){
  var aoi = drawingTools.layers().get(0).getEeObject();
  var lato = Side.getValue()*1000;
  var square = aoi.buffer(ee.Number(Side.getValue()*1000).divide(2), 1).bounds().transform('EPSG:3857', 1);
  var cliproi = ((dem)
  .reduceResolution({
    reducer: ee.Reducer.mean(),
    maxPixels: 3840 
    // range 126 km max 
    //1024 range 250 km max
  }).reproject({
      crs: 'EPSG:3857',
      scale: 60
    })).visualize(vis_parms).clip(square);
//  .reproject('EPSG:3857').clip(square));
  Map.addLayer(square, null, 'point');
  Map.centerObject(square);
  return cliproi;
}
